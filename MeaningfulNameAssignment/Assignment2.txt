Assignment - 2 The below program is to guess the correct number between 1 to 100

def isValidInput(number):
    if number.isdigit() and 1<= int(number) <=100:
        return True
    else:
        return False

def main():
    targetNumber=random.randint(1,100)
    result=False
    userInput=input("Guess a number between 1 and 100:")
    validCounter=0
    while not result:
        if not isValidInput(userInput):
            userInput=input("I wont count this one Please enter a number between 1 to 100")
            continue
        else:
            validCounter+=1
            userInput=int(userInput)

        if userInput<targetNumber:
            userInput=input("Too low. Guess again")
        elif userInput>targetNumber:
            userInput=input("Too High. Guess again")
        else:
            print("You guessed it in",validCounter,"guesses!")
            result=True
main()
